cmake_minimum_required(VERSION 3.20)
project(poppler_goofile_windows_validation LANGUAGES CXX)

enable_testing()

set(POPPLER_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/third_party/poppler")
set(POPPLER_CONFIG_DIR "${CMAKE_CURRENT_SOURCE_DIR}/generated")
set(POPPLER_BASELINE_DIR "${CMAKE_CURRENT_BINARY_DIR}/baseline-poppler")
set(POPPLER_CANDIDATE_DIR "${CMAKE_CURRENT_BINARY_DIR}/candidate-poppler")

file(MAKE_DIRECTORY "${POPPLER_BASELINE_DIR}/goo" "${POPPLER_CANDIDATE_DIR}/goo")
configure_file("${POPPLER_SOURCE_DIR}/goo/gfile.h" "${POPPLER_BASELINE_DIR}/goo/gfile.h" COPYONLY)
configure_file("${POPPLER_SOURCE_DIR}/goo/gfile.cc" "${POPPLER_BASELINE_DIR}/goo/gfile.cc" COPYONLY)
configure_file("${POPPLER_SOURCE_DIR}/goo/gfile.h" "${POPPLER_CANDIDATE_DIR}/goo/gfile.h" COPYONLY)

file(READ "${POPPLER_SOURCE_DIR}/goo/gfile.cc" candidate_source)
set(original_sharing "FILE_SHARE_READ | FILE_SHARE_WRITE, nullptr, OPEN_EXISTING, FILE_ATTRIBUTE_NORMAL, nullptr")
set(candidate_sharing "FILE_SHARE_READ | FILE_SHARE_WRITE | FILE_SHARE_DELETE, nullptr, OPEN_EXISTING, FILE_ATTRIBUTE_NORMAL, nullptr")
string(REGEX MATCHALL "FILE_SHARE_READ \\| FILE_SHARE_WRITE, nullptr, OPEN_EXISTING, FILE_ATTRIBUTE_NORMAL, nullptr" original_matches "${candidate_source}")
list(LENGTH original_matches original_match_count)
if(NOT original_match_count EQUAL 2)
  message(FATAL_ERROR "Expected two unpatched GooFile sharing expressions, found ${original_match_count}")
endif()
string(REPLACE "${original_sharing}" "${candidate_sharing}" candidate_source "${candidate_source}")
string(REGEX MATCHALL "FILE_SHARE_DELETE" delete_matches "${candidate_source}")
list(LENGTH delete_matches delete_match_count)
if(NOT delete_match_count EQUAL 2)
  message(FATAL_ERROR "Expected two FILE_SHARE_DELETE expressions after patching, found ${delete_match_count}")
endif()
file(WRITE "${POPPLER_CANDIDATE_DIR}/goo/gfile.cc" "${candidate_source}")

function(add_goofile_delete_recreate_test variant source_dir expected_delete_success)
  add_library(gfile-${variant} STATIC "${source_dir}/goo/gfile.cc")
  target_include_directories(gfile-${variant} PRIVATE "${POPPLER_CONFIG_DIR}")
  target_compile_features(gfile-${variant} PRIVATE cxx_std_20)

  add_executable(goo-file-${variant}-delete-recreate goo_file_delete_recreate_test.cpp)
  target_include_directories(goo-file-${variant}-delete-recreate PRIVATE "${source_dir}/goo" "${POPPLER_CONFIG_DIR}")
  target_compile_features(goo-file-${variant}-delete-recreate PRIVATE cxx_std_20)
  target_compile_definitions(goo-file-${variant}-delete-recreate PRIVATE EXPECT_DELETE_SUCCESS=${expected_delete_success})
  target_link_libraries(goo-file-${variant}-delete-recreate PRIVATE gfile-${variant})
  add_test(NAME goo-file-${variant}-delete-recreate COMMAND goo-file-${variant}-delete-recreate)
endfunction()

add_goofile_delete_recreate_test(baseline "${POPPLER_BASELINE_DIR}" 0)
add_goofile_delete_recreate_test(candidate "${POPPLER_CANDIDATE_DIR}" 1)

add_executable(windows-file-sharing-diagnostics windows_file_sharing_test.cpp)
target_compile_features(windows-file-sharing-diagnostics PRIVATE cxx_std_20)
add_test(NAME windows-file-sharing-diagnostics COMMAND windows-file-sharing-diagnostics)
